# Helper to encode logo image for clean HTML rendering
def get_base64_image(image_path):
    if os.path.exists(image_path):
        with open(image_path, "rb") as img_file:
            return base64.b64encode(img_file.read()).decode("utf-8")
    return ""

# 4. CORE ENGINE & PIPELINE

tab_synth, tab_library = st.tabs(["⚡ Convert Document", "📚 Personal Library"])

def extract_text_from_pdf(pdf_file) -> str:
    reader = PdfReader(pdf_file)
    extracted = ""
    for page in reader.pages:
        text = page.extract_text()
        if text:
            extracted += text + "\n"
    return extracted.strip()

with tab_synth:
    uploaded_file = st.file_uploader("Drop document", type=["pdf"], label_visibility="collapsed")

    if uploaded_file is not None:
        st.caption(f"Attached: **{uploaded_file.name}** ({round(uploaded_file.size / 1024, 1)} KB)")
        
        if st.button("Synthesize Audio 🎧"):
            with st.status("Processing document...", expanded=True) as status:
                st.write("📄 Extracting text...")
                raw_text = extract_text_from_pdf(uploaded_file)
                
                if not raw_text:
                    st.error("No readable text found in document.")
                    st.stop()
                
                input_text = raw_text[:4000]
                
                st.write("🎙️ Generating voice stream...")
                audio_generator = eleven_client.text_to_speech.convert(
                    text=input_text,
                    voice_id="JBFqnCBsd6RMkjVDRZzb",
                    model_id="eleven_flash_v2_5"
                )
                audio_bytes = b"".join(audio_generator)
                
                st.write("☁️ Uploading to cloud storage...")
                clean_filename = uploaded_file.name.replace(".pdf", ".mp3").replace(" ", "_")
                storage_path = f"public/{uuid.uuid4()}_{clean_filename}"
                
                supabase.storage.from_("audiobooks").upload(
                    path=storage_path,
                    file=audio_bytes,
                    file_options={"content-type": "audio/mpeg"}
                )
                
                public_audio_url = supabase.storage.from_("audiobooks").get_public_url(storage_path)
                
                supabase.table("audiobooks").insert({
                    "user_id": user_id,
                    "title": uploaded_file.name,
                    "audio_file_url": public_audio_url,
                    "file_size_bytes": len(audio_bytes)
                }).execute()
                
                status.update(label="Complete!", state="complete", expanded=False)
            
            st.subheader("Output")
            st.audio(public_audio_url, format="audio/mp3")

with tab_library:
    st.subheader("Saved Audiobooks")
    try:
        records = supabase.table("audiobooks").select("*").eq("user_id", user_id).order("created_at", desc=True).execute()
        if not records.data:
            st.info("No audiobooks generated yet.")
        else:
